\subsection{Vấn đề té ngã và tình hình nghiên cứu}

\begin{frame}{Nghiên cứu trong và ngoài nước}
\begin{columns}[T]
    \begin{column}{0.48\textwidth}
        \textbf{Quốc tế}
        \begin{itemize}
            \item Xu hướng: sử dụng AI nhẹ, YOLO, Transformer, cảm biến mmWave
            \item Thành tựu: giảm false alarm, tối ưu cho thiết bị biên, kết hợp dữ liệu từ nhiều cảm biến (Sensor Fusion)
        \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
        \textbf{Trong nước}
        \begin{itemize}
            \item Thực trạng: chủ yếu mô hình thử nghiệm (PoC) với ESP32, Arduino
            \item Hạn chế: thiếu dữ liệu lớn, độ chính xác thấp (75-85\%), thiếu tích hợp đa phương thức
        \end{itemize}
    \end{column}
\end{columns}
\end{frame}
