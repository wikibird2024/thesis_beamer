
\subsection{Mục tiêu đặt ra và phạm vi thực hiện của đề tài} 

% --- Slide 10: Hiệu năng và giới hạn ---
\begin{frame}{Mục tiêu và giới hạn của sản phẩm}
    \begin{columns}[T,onlytextwidth]
        % --- Cột 1: Mục tiêu ---
        \begin{column}{0.48\textwidth}
            \begin{block}{\textcolor{blue!70!black}{\textbf{Mục tiêu hiệu năng}}}
                \begin{itemize}
                    \item[$\checkmark$] Tổng độ trễ $<5$ giây.
                    \item[$\checkmark$] Độ chính xác $>90\%$, False Alarm $<8\%$.
                    \item[$\checkmark$] Uptime dịch vụ MQTT $>99\%$.
                \end{itemize}
            \end{block}

            \begin{block}{\textcolor{teal!70!black}{\textbf{Lựa chọn đa phương thức}}}
                Kết hợp dữ liệu từ nhiều cảm biến giúp hệ thống giám sát rộng hơn 
                và nhận diện té ngã linh hoạt hơn.
            \end{block}
        \end{column}

        % --- Cột 2: Giới hạn ---
        \begin{column}{0.48\textwidth}
            \begin{block}{\textcolor{red!70!black}{\textbf{Giới hạn sản phẩm}}}
                \begin{itemize}
                    \item[$\times$] Hoạt động trong nhà với điều kiện ánh sáng và mạng ổn định.
                    \item[$\times$] Nguyên mẫu chưa tích hợp học sâu toàn phần.
                    \item[$\times$] Không phát triển app di động/web phức tạp.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}
