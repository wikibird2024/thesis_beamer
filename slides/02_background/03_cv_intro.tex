\subsection{Thị giác máy tính và nhận diện tư thế người}

% ---------- Slide 1 ----------
\begin{frame}
\frametitle{Định nghĩa và ứng dụng của computer vision}

\begin{block}{Định nghĩa}
\textbf{Thị giác Máy tính (CV)}: Lĩnh vực AI cho phép máy tính xử lý, phân tích và diễn giải hình ảnh/video, mô phỏng thị giác con người.
\end{block}

\begin{block}{Mục tiêu}
\begin{itemize}
    \item Tái tạo khả năng nhận thức thị giác với tốc độ, độ chính xác và quy mô vượt trội.
    \item Ứng dụng trong \TENLUANVAN, đặc biệt là \textbf{Ước lượng Tư thế Người (HPE)}.
\end{itemize}
\end{block}
\end{frame}

% ---------- Slide 2 ----------
\begin{frame}
\frametitle{Pipeline Cơ bản của Hệ thống CV}

\begin{block}{Quy trình}
\begin{enumerate}
    \item \textbf{Thu nhận dữ liệu}: Thu thập ảnh/video từ camera.
    \item \textbf{Tiền xử lý}: Chuẩn hóa kích thước, điều chỉnh sáng/tương phản, giảm nhiễu.
    \item \textbf{Trích xuất đặc trưng}: Chuyển pixel thành đặc trưng trừu tượng (cạnh, góc, kết cấu).
    \item \textbf{Phân tích và quyết định}: Phân loại, nhận dạng hoặc ước lượng tư thế.
\end{enumerate}
\end{block}

\begin{frame}
\frametitle{Mô hình Học sâu: CNN (Convolution Neural Network)}

\begin{columns}
% --- Cột trái ---
\begin{column}{0.5\textwidth}
\begin{itemize}
    \item \textbf{Mạng Nơ-ron Tích chập (CNN)}: Kiến trúc chủ đạo cho xử lý ảnh.
    \item \textbf{Phép tích chập}: Trích xuất đặc trưng cục bộ từ vùng nhỏ của ảnh.
    \begin{equation}
    \footnotesize
    (I * K)(i, j) = \sum_{m} \sum_{n} I(i-m, j-n) \, K(m, n)
    \end{equation}
    \begin{itemize}
        \item $I$: ma trận ảnh đầu vào.
        \item $K$: kernel (bộ lọc).
        \item $(i,j)$: vị trí pixel trong đầu ra.
    \end{itemize}
    \item \textbf{Phép gộp (Pooling)}: Giảm kích thước, tăng tính bền vững.
\end{itemize}
\end{column}

% ---------- Slide ViT ----------
\begin{frame}
\frametitle{Mô hình Học sâu: ViT (Vision Transformer)}

\begin{columns}
% --- Cột trái ---
\begin{column}{0.5\textwidth}
\begin{itemize}
    \item \textbf{ViT}: Chia ảnh thành \textbf{patch} (miếng vá), xử lý như chuỗi token.
    \item \textbf{Cơ chế Self-Attention}: Tính trọng số quan hệ giữa các token.
    \begin{equation}
    \footnotesize
    \text{Attention}(Q,K,V) =
    \text{softmax}\!\left(\frac{QK^T}{\sqrt{d_k}}\right) V
    \end{equation}
    \begin{itemize}
        \item $Q,K,V$: ma trận truy vấn, khóa, giá trị.
        \item $d_k$: độ dài vector khóa (chuẩn hóa).
    \end{itemize}
    \item Giúp học \textbf{quan hệ toàn cục}, vượt giới hạn cục bộ của CNN.
\end{itemize}
\end{column}

\begin{frame}
\frametitle{Các Bài toán Chính trong CV}

\begin{exampleblock}{Các nhóm bài toán thường gặp}
\begin{itemize}
    \item \textbf{Phân loại Ảnh (Image Classification)}  
    → Đầu ra: một nhãn duy nhất cho cả bức ảnh.  
    \textit{Ví dụ:} ảnh chứa người hay xe.
    
    \item \textbf{Phát hiện Đối tượng (Object Detection)}  
    → Đầu ra: nhãn + vị trí (hộp giới hạn bounding box).  
    \textit{Ví dụ:} phát hiện nhiều người trong cùng một khung hình.
    
    \item \textbf{Phân đoạn Ảnh (Image Segmentation)}  
    \begin{itemize}
        \item \textbf{Ngữ nghĩa (Semantic)}: gán nhãn từng pixel theo lớp.  
        \textit{Ví dụ:} tất cả pixel thuộc “Đường”, “Cây”.
        \item \textbf{Thể hiện (Instance)}: phân biệt từng cá thể trong cùng lớp.  
        \textit{Ví dụ:} tách riêng từng chiếc xe thay vì gom tất cả thành “Xe”.
    \end{itemize}
    
    \item \textbf{\textcolor{orange}{Ước lượng Tư thế Người (Human Pose Estimation - HPE)}}
    → Đầu ra: tọa độ \textbf{các khớp xương (keypoints)} của cơ thể.  
    \textit{Ứng dụng:} phân tích chuyển động, giám sát té ngã.
\end{itemize}
\end{block}
\end{frame}
