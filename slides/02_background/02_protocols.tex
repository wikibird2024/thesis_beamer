
\subsection{Các phương thức truyền thông sử dụng}

% =====================================================
% Slide: Giao thức SIP - Chức năng và Đường tín hiệu/Media
% =====================================================
\begin{frame}{Giao thức SIP - Khởi tạo Phiên và Đường tín hiệu/Media}

% --- Chức năng ---
\begin{block}{\textbf{Chức năng chính của SIP trong hệ thống}}
\begin{itemize}
    \item Thiết lập và quản lý cuộc gọi VoIP từ hệ thống cảnh báo.
    \item Kết nối với \textbf{Asterisk PBX} để gọi điện thoại.
    \item Truyền âm thanh cảnh báo qua \textbf{RTP}.
\end{itemize}
\end{block}

% --- Quy trình hoạt động ---
\begin{block}{\textbf{Quy trình hoạt động cơ bản}}
    \textcolor{blue!70!black}{REGISTER} $\;\rightarrow\;$ 
    \textcolor{green!60!black}{INVITE} $\;\rightarrow\;$ 
    \textcolor{orange!80!black}{ACK} $\;\rightarrow\;$ 
    \textcolor{purple!70!black}{RTP} $\;\rightarrow\;$ 
    \textcolor{red!70!black}{BYE}
\end{block}

% --- Cột: Signaling vs Media ---
\begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \begin{block}{\textcolor{blue!70!black}{Đường tín hiệu (Signaling Path)}}
        \begin{itemize}
            \item Mang bản tin SIP: \texttt{INVITE},\texttt{BYE}, \dots
            \item Thiết lập, quản lý, kết thúc cuộc gọi.
            \item Chạy trên \textbf{TCP/UDP}.
        \end{itemize}
    \end{block}

    \column{0.48\textwidth}
    \begin{block}{\textcolor{purple!70!black}{Đường truyền phương tiện (Media Path)}}
        \begin{itemize}
            \item Mang dữ liệu thoại/video thực tế.
            \item Sử dụng \textbf{RTP qua UDP}.
            \item Truyền trực tiếp giữa các điểm cuối.
        \end{itemize}
    \end{block}
\end{columns}
\end{frame}

\begin{frame}{SIP – ICE và Asterisk}
\begin{columns}[T, totalwidth=\textwidth]
    % --- Cột 1: NAT và ICE ---
    \begin{column}{0.5\textwidth}
        \begin{block}{\textbf{NAT / Firewall}}
            \begin{itemize}
                \item Thiết bị VoIP thường ở sau NAT/Firewall.

        \begin{block}{\textbf{ICE (Interactive Connectivity Establishment)}}
            \begin{itemize}
                \item \textbf{Local IP}: địa chỉ nội bộ. 
                \item \textbf{STUN}: phát hiện IP công cộng/cổng NAT.
                \item \textbf{TURN}: chuyển tiếp media khi STUN thất bại.
            \end{itemize}
        \end{block}
    \end{column}

    % --- Cột 2: Asterisk ---
    \begin{column}{0.46\textwidth}

                \item Quản lý và định tuyến cuộc gọi VoIP.

\begin{frame}[fragile]
\frametitle{MQTT: Tổng quan và Kiến trúc Pub/Sub}

\begin{block}{Định nghĩa \& Đặc điểm}
\begin{itemize}
  \item \textbf{MQTT}: Message Queuing Telemetry Transport
  \item Giao thức \textbf{nhẹ}, tối ưu cho IoT/M2M
  \item Hoạt động trên \textbf{TCP/IP}, kết nối lâu dài
  \item Thích hợp cho thiết bị \textbf{tài nguyên hạn chế}, mạng \textbf{băng thông thấp}
\end{itemize}
\end{block}

\begin{block}{Kiến trúc Publish/Subscribe}
\centering
\resizebox{0.8\textwidth}{!}{%
\begin{tikzpicture}[node distance=1.2cm,>=stealth,thick]
  % Broker
  \node[draw, rectangle, rounded corners, fill=blue!20, minimum width=2cm, minimum height=1cm] (broker) {Broker};

  % Publishers
  \node[draw, ellipse, fill=green!20, left=2.5cm of broker, yshift=1.0cm] (pub1) {Publisher 1};
  \node[draw, ellipse, fill=green!20, left=2.5cm of broker, yshift=-1.0cm] (pub2) {Publisher 2};

  % Subscribers
  \node[draw, ellipse, fill=orange!20, right=2.5cm of broker, yshift=1.0cm] (sub1) {Subscriber A};
  \node[draw, ellipse, fill=orange!20, right=2.5cm of broker, yshift=-1.0cm] (sub2) {Subscriber B};

  % Arrows
  \draw[->] (pub1) -- (broker) node[midway, above] {\tiny publish};
  \draw[->] (pub2) -- (broker) node[midway, below] {\tiny publish};

  \draw[->] (broker) -- (sub2) node[midway, below] {\tiny notify};
\end{tikzpicture}
}

\vspace{0.3cm}
{\scriptsize Ví dụ topic: \texttt{sensor/room/temperature}, \texttt{alert/fall/detected}}
\end{block}

\end{frame}

% Slide 4: MQTT - QoS và Bảo mật
% =====================================================
\begin{frame}{MQTT: QoS, Lợi ích và Bảo mật}
\small
\begin{columns}[T,totalwidth=\textwidth]
    % --- Cột 1: QoS & Lợi ích ---
    \begin{column}{0.48\textwidth}
        \begin{block}{\textbf{QoS Levels}}
            \begin{itemize}
                \item \textcolor{green!60!black}{\textbf{QoS 0}} – Fire \& Forget, nhanh, có thể mất bản tin.
                \item \textcolor{orange!80!black}{\textbf{QoS 1}} – At least once, đảm bảo nhận, có thể trùng.
                \item \textcolor{red!70!black}{\textbf{QoS 2}} – Exactly once, tin cậy nhất nhưng chậm.
            \end{itemize}
        \end{block}

        \begin{block}{\textbf{Lợi ích}}
            \begin{itemize}
                \item Truyền/nhận tách biệt (Pub–Sub).
                \item Giảm độ trễ, hỗ trợ offline.
                \item Dễ mở rộng hệ thống.
            \end{itemize}
        \end{block}
    \end{column}

    % --- Cột 2: Bảo mật ---
    \begin{column}{0.48\textwidth}
        \begin{block}{\textbf{Bảo mật trong MQTT}}
            \begin{itemize}
                \item \textbf{Mã hoá kênh}: TLS/SSL (\texttt{port 8883}), certificate pinning.
                \item \textbf{Xác thực}: Username/Password, Client certificates, Token (JWT, OAuth2).
                \item \textbf{Uỷ quyền}: ACL theo topic, RBAC cho nhóm người dùng.
                \item \textbf{Bảo vệ hệ thống}: hạn chế kết nối, rate limiting, chống DoS.
            \end{itemize}
        \end{block}
    \end{column}
\end{columns}
\end{frame}

\begin{frame}[fragile]
\frametitle{JSON - Định nghĩa, Cấu trúc và Thư viện}

\begin{columns}[T]
\column{0.48\textwidth}
\textbf{Định nghĩa & cấu trúc:}
\begin{itemize}
\item Định dạng nhẹ, độc lập ngôn ngữ
\item Key:value, Object, Array, String, Number, Boolean, null
\item Ứng dụng: lưu cấu hình, payload MQTT, API, cảnh báo
\end{itemize}

\textbf{Thư viện nhúng:}
\begin{itemize}
\item \texttt{json-c}: C cho C/C++
\item \texttt{FirebaseJson}: dễ dùng cho IoT
\end{itemize}

\column{0.48\textwidth}
\begin{exampleblock}{Ví dụ JSON}
\begin{minted}[frame=single, fontsize=\tiny, breaklines]{json}
{
  "device_id": "ESP32_001",
  "temperature": 25.5,
  "sensors": ["temp", "light"]
}
\end{minted}
\end{exampleblock}
\end{columns}

\end{frame}

\begin{frame}
\frametitle{Tối Ưu Hóa và Kết Hợp SIP/MQTT/JSON}

\begin{columns}[T,onlytextwidth]
    % ---- Cột 1: Vai trò ----
    \begin{column}{0.48\textwidth}
        \textbf{Vai trò từng giao thức:}
        \begin{itemize}
            \item \textcolor{blue}{\textbf{MQTT}}: Truyền dữ liệu IoT hiệu quả, hỗ trợ QoS.
            \item \textcolor{teal}{\textbf{JSON}}: Định dạng linh hoạt, dễ tích hợp hệ thống.
            \item \textcolor{red}{\textbf{SIP}}: Cảnh báo âm thanh/thoại tức thì.
        \end{itemize}
    \end{column}

    % ---- Cột 2: Tối ưu & Bảo mật ----
    \begin{column}{0.48\textwidth}
        \textbf{Hướng tối ưu:}
        \begin{itemize}
            \item Payload JSON nhỏ gọn, giảm băng thông.
            \item Chọn mức QoS phù hợp cho từng loại dữ liệu.
            \item Tự động reconnect, đảm bảo tính sẵn sàng.
            \item \textbf{Bảo mật}: TLS + xác thực người dùng.
        \end{itemize}
    \end{column}
\end{columns}

\vspace{0.5em}
\begin{block}{\centering Tích hợp đa giao thức = \textbf{Hiệu quả + Độ tin cậy + An toàn}}
\end{block}

\end{frame}
