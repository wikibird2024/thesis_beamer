\begin{frame}{Giao thức SIP - Khởi tạo Phiên và Đường tín hiệu/Media}

% --- Chức năng ---
\begin{block}{\textbf{Chức năng chính của SIP trong hệ thống}}
\begin{itemize}
    \item Thiết lập và quản lý cuộc gọi VoIP từ hệ thống cảnh báo.
    \item Kết nối với \textbf{Asterisk PBX} để gọi điện thoại.
    \item Truyền âm thanh cảnh báo qua \textbf{RTP}.
\end{itemize}
\end{block}

% --- Quy trình hoạt động ---
\begin{block}{\textbf{Quy trình hoạt động cơ bản}}
    \textcolor{blue!70!black}{REGISTER} $\;\rightarrow\;$ 
    \textcolor{green!60!black}{INVITE} $\;\rightarrow\;$ 
    \textcolor{orange!80!black}{ACK} $\;\rightarrow\;$ 
    \textcolor{purple!70!black}{RTP} $\;\rightarrow\;$ 
    \textcolor{red!70!black}{BYE}
\end{block}

% --- Cột: Signaling vs Media ---
\begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \begin{block}{\textcolor{blue!70!black}{Đường tín hiệu (Signaling Path)}}
        \begin{itemize}
            \item Mang bản tin SIP: \texttt{INVITE},\texttt{BYE}, \dots
            \item Thiết lập, quản lý, kết thúc cuộc gọi.
            \item Chạy trên \textbf{TCP/UDP}.
        \end{itemize}
    \end{block}

    \column{0.48\textwidth}
    \begin{block}{\textcolor{purple!70!black}{Đường truyền phương tiện (Media Path)}}
        \begin{itemize}
            \item Mang dữ liệu thoại/video thực tế.
            \item Sử dụng \textbf{RTP qua UDP}.
            \item Truyền trực tiếp giữa các điểm cuối.
        \end{itemize}
    \end{block}
\end{columns}
\end{frame}

% Slide 2: SIP - ICE và Asterisk
% =====================================================
\begin{frame}
\frametitle{SIP - ICE và Asterisk}

\begin{block}{Vấn đề NAT/Firewall}
Các thiết bị thường nằm sau NAT/firewall, ngăn cản truyền RTP trực tiếp
\end{block}

\begin{block}{ICE}
\begin{itemize}
\item Local IP: địa chỉ nội bộ
\item STUN: phát hiện IP công cộng/cổng NAT
\item TURN: máy chủ chuyển tiếp khi STUN thất bại
\end{itemize}
\end{block}

\begin{block}{Asterisk}
\begin{itemize}
\item Quản lý tập trung, chuẩn mở, bảo mật TLS/SRTP
\item Xử lý đăng ký và định tuyến cuộc gọi
\end{itemize}
\end{block}
\end{frame}

% =====================================================
\begin{frame}[fragile]
\frametitle{MQTT - Tổng quan và Kiến trúc Publish/Subscribe}

\begin{columns}[T]
\column{0.5\textwidth}
\textbf{Định nghĩa & đặc điểm:}
\begin{itemize}
\item MQTT = Message Queuing Telemetry Transport
\item Giao thức nhẹ, tối ưu IoT/M2M
\item TCP/IP, kết nối lâu dài
\item Phù hợp thiết bị tài nguyên hạn chế, băng thông thấp
\end{itemize}

\column{0.5\textwidth}
\textbf{Kiến trúc Pub/Sub:}
\begin{itemize}
\item Broker trung tâm: Mosquitto/HiveMQ
\item Publishers gửi dữ liệu, Subscribers nhận thông báo
\item Ví dụ topic: \texttt{sensor/room/temperature}, \texttt{alert/fall/detected}
\end{itemize}
\end{columns}

% (TikZ sơ đồ gọn)
\end{frame}

% =====================================================
% Slide 4: MQTT - QoS và Bảo mật
% =====================================================
\begin{frame}
\frametitle{MQTT - QoS, Lợi ích và Bảo mật}

\begin{columns}[T]
\column{0.5\textwidth}
\textbf{QoS:}
\begin{itemize}
\item QoS 0: Fire & Forget, nhanh, có thể mất message
\item QoS 1: Gửi ít nhất 1 lần, có thể trùng lặp
\item QoS 2: Gửi đúng 1 lần, tin cậy nhất, chậm
\end{itemize}

\textbf{Lợi ích:} Truyền/nhận độc lập, giảm độ trễ, hỗ trợ offline, scale tốt

\column{0.5\textwidth}
\textbf{Bảo mật:}
\begin{itemize}
\item TLS/SSL, MQTT over TLS (port 8883)
\item Username/Password, client certificates
\item ACL kiểm soát quyền truy cập topic
\end{itemize}
\end{columns}

\end{frame}

\begin{frame}[fragile]
\frametitle{JSON - Định nghĩa, Cấu trúc và Thư viện}

\begin{columns}[T]
\column{0.48\textwidth}
\textbf{Định nghĩa & cấu trúc:}
\begin{itemize}
\item Định dạng nhẹ, độc lập ngôn ngữ
\item Key:value, Object, Array, String, Number, Boolean, null
\item Ứng dụng: lưu cấu hình, payload MQTT, API, cảnh báo
\end{itemize}

\textbf{Thư viện nhúng:}
\begin{itemize}
\item \texttt{json-c}: C cho C/C++
\item \texttt{FirebaseJson}: dễ dùng cho IoT
\end{itemize}

\column{0.48\textwidth}
\begin{exampleblock}{Ví dụ JSON}
\begin{minted}[frame=single, fontsize=\tiny, breaklines]{json}
{
  "device_id": "ESP32_001",
  "temperature": 25.5,
  "sensors": ["temp", "light"]
}
\end{minted}
\end{exampleblock}
\end{columns}

\end{frame}
