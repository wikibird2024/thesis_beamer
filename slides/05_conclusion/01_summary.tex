% Slide 1: Kết quả & Hướng phát triển
% ----------------------------
\begin{frame}{Kết quả chung \& Hướng phát triển}
\begin{columns}[T]
    % Column 1: Kết quả
    \column{0.5\textwidth}
    \textbf{Kết quả chính}
    \begin{itemize}
        \item Hệ thống IoT tích hợp hoàn chỉnh từ cảm biến - xử lý - thiết bị cuối
        \item Thuật toán phát hiện té ngã: kết hợp dữ liệu cảm biến và hình ảnh theo thời gian thực.
        \item Cơ chế cảnh báo đa dạng: buzzer, LED, SMS, MQTT/Telegram, SIP.
        \item Thực nghiệm: hệ thống hoạt động với độ chính xác tương đối, phản hồi nhanh 
        \item Tiềm năng ứng dụng: giám sát sức khỏe từ xa, ứng dụng trong các trung tâm chăm sóc sức khỏe.
    \end{itemize}

    % Column 2: Hạn chế & hướng phát triển
    \column{0.5\textwidth}
    \textbf{Hạn chế \& Hướng phát triển}
    \begin{itemize}
        \item Quy mô thử nghiệm còn hạn chế; số lượng thiết bị đầu cuối còn hạn chế
        \item Thuật toán chưa hoàn toàn ổn định; thiết bị đeo hiện tại còn cồng kềnh.
        \item Kết nối 4G/GPS chưa tối ưu, độ tin cậy cần cải thiện.
        \item \textbf{Hướng phát triển}: mở rộng quy mô thử nghiệm, cải tiến thuật toán, thu gọn kích thước thiết bị, hoàn thiện kết nối 4G/GPS, tích hợp nền tảng IoT.
    \end{itemize}
\end{columns}
\end{frame}

    \item Xác định các thách thức thực tế: tối ưu phần cứng, nâng cao độ chính xác thuật toán, đảm bảo truyền thông ổn định.
\end{itemize}
