\textbf{PJSIP} \newline

\textit{(Kịch bản xử lý)} &

\hline
\textbf{AMI} \newline
\textit{(Giao diện quản lý)} & 
Cho phép ứng dụng Python điều khiển Asterisk từ xa (gọi, ngắt máy, lấy trạng thái) &
Kết nối qua mạng nội bộ với xác thực username/password \\
\hline

\begin{columns}[t]
\begin{column}{0.6\textwidth}
\begin{minted}[fontsize=\tiny, frame=single, breaklines]{ini}
[internal]
exten => 6001,1,Answer()
same => n,Dial(PJSIP/6001,20)
same => n,Hangup()

exten => 6000,1,Dial(PJSIP/6001&PJSIP/6002&PJSIP/6003,20)
same => n,Hangup()

[messages]
exten => _X.,1,NoOp(===> SIP MESSAGE from ${MESSAGE(from)})
same => n,MessageSend(pjsip:${EXTEN},pjsip:server)
same => n,NoOp(===> Send status: ${MESSAGE_SEND_STATUS})
same => n,Hangup()
\end{minted}
\end{column}

\item Extension 6000: gọi nhóm
\end{itemize}
\end{block}
