
\begin{frame}[fragile]{Module Camera: Tổng Quan \& Luồng Hoạt Động}
    \small
    \begin{columns}[T, onlytextwidth]
        % ────── Cột 1: Tổng quan ──────
        \begin{column}{0.48\textwidth}
            \begin{block}{Tổng quan}
                \begin{itemize}
                    \item Vi điều khiển \textbf{ESP32-S3} với camera \textbf{OV5640}
                    \item Truyền video MJPEG qua HTTP thời gian thực
                    \item Kiến trúc hướng sự kiện (\textbf{ESP-IDF}) quản lý camera, mạng và bộ đệm
                \end{itemize}
            \end{block}

            \begin{alertblock}{Đặc điểm kỹ thuật}
                Triple buffering trên PSRAM giảm độ trễ
            \end{alertblock}
        \end{column}

        % ────── Cột 2: Luồng hoạt động ──────
        \begin{column}{0.48\textwidth}
            \begin{block}{Luồng hoạt động}
                \begin{enumerate}
                    \item \textbf{Khởi tạo}: Cấu hình Wi-Fi, camera, HTTP server
                    \item \textbf{Chờ kết nối}: Lắng nghe yêu cầu client
                    \item \textbf{Lấy frame}: Đọc khung hình từ buffer
                    \item \textbf{Truyền tải}: Gửi JPEG qua HTTP multipart
                    \item \textbf{Giám sát}: Theo dõi FPS và xử lý lỗi
                \end{enumerate}
            \end{block}
        \end{column}
    \end{columns}
